# The toolchain the shipped binaries are built with.

# musl rather than glibc because the agent is copied to someone else's host and
# has to run there: statically linked glibc still resolves names through NSS,
# which needs a matching glibc present at runtime. musl has no such split.
#
# One package per line. The hash of this file keys the package cache in CI, so
# an edit here refreshes the cache by construction.

lld
llvm20
compiler-rt
libc++-static
libc++-dev
llvm-libunwind-static
musl-dev
cmake
ninja
ccache
bash
git

# a separate reviewed commit, not something that arrives silently with the base
# image. 1.0.20-r1 carries Alpine's patch for CVE-2025-69277; it is built without
# LTO (invariant 6a), and the distribution runs make check against it.
libsodium-static=1.0.20-r1
